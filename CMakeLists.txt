cmake_minimum_required(VERSION 3.20)
project(VehicleSensorThreadpool VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(ENABLE_TSAN "Enable ThreadSanitizer (supported Linux compilers only)" OFF)
option(ENABLE_OPENMP "Build the optional OpenMP library baseline" ON)
add_library(vehicle_core src/sensor/SensorSimulator.cpp src/sensor/SensorProcessor.cpp src/sensor/DatasetExport.cpp src/threadpool/Executors.cpp src/threadpool/LockFreePool.cpp src/benchmark/Metrics.cpp src/benchmark/ProcessProfile.cpp)
target_include_directories(vehicle_core PUBLIC include)
target_link_libraries(vehicle_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(vehicle_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(vehicle_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ENABLE_TSAN)
  if(WIN32 OR MSVC)
    message(FATAL_ERROR "ThreadSanitizer requires a supported Linux GCC/Clang toolchain")
  endif()
  target_compile_options(vehicle_core PUBLIC -fsanitize=thread -g)
  target_link_options(vehicle_core PUBLIC -fsanitize=thread)
endif()
if(ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(vehicle_core PUBLIC OpenMP::OpenMP_CXX)
    target_compile_definitions(vehicle_core PUBLIC VEHICLE_OPENMP=1)
  endif()
endif()
add_executable(vehicle_benchmark src/main.cpp src/Menu.cpp)
target_link_libraries(vehicle_benchmark PRIVATE vehicle_core)
enable_testing()
add_executable(test_threadpool tests/test_threadpool.cpp)
target_link_libraries(test_threadpool PRIVATE vehicle_core)
add_test(NAME correctness_and_stress COMMAND test_threadpool)
set_tests_properties(correctness_and_stress PROPERTIES TIMEOUT 120)
add_test(NAME cli_smoke COMMAND vehicle_benchmark --scheduler all --tasks 80 --threads 2 --repeats 1 --warmups 0 --lidar-points 128 --output smoke/results.csv)
set_tests_properties(cli_smoke PROPERTIES TIMEOUT 60)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_exports_and_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:vehicle_benchmark>")
  set_tests_properties(cli_exports_and_validation PROPERTIES TIMEOUT 60)
endif()
